#include <iostream>
#include <cmath>

using namespace std;

const int N = 110;
const double eps = 1e-8;

int n, t;
double a[N][N];

int guass()
{
	int c, r;
	for (c = 0, r = 0; c < n; c ++ )
	{
		int t = r;
		for (int i = r + 1; i < n; i ++ )
			if (fabs(a[t][c]) < fabs(a[i][c])) t = i;
		if (fabs(a[t][c]) < eps) continue;
		for (int i = c; i <= n; i ++ ) swap(a[t][i], a[r][i]);
		
		for (int i = n; i >= c; i -- ) a[r][i] /= a[r][c];
		
		for (int i = r + 1; i < n; i ++ )
			if (fabs(a[i][c]) > eps)
				for (int j = n; j >= c; j -- )
					a[i][j] -= a[i][c] * a[r][j];
		r ++ ;
	}
	if (r < n)
	{
		for (int i = r + 1; i < n; i ++ ) if (fabs(a[i][n]) > eps) return 2; //方程组无解
		return 1; //方程组有无穷多组解
	}
	for (int i = n - 1; i >= 0; i -- )
		for (int j = i + 1; j < n; j ++ )
			a[i][n] -= a[i][j] * a[j][n];
	return 0;
}

int main()
{
	cin >> n;
	for (int i = 0; i < n; i ++ )
		for (int j = 0; j <= n; j ++ )
			cin >> a[i][j];
	t = guass();
	if (t == 1 || t == 2) cout << "No Solution" << endl;
	else
	{
		for (int i = 0; i < n; i ++ )
		{
			if (fabs(a[i][n]) < eps) a[i][n] = 0;
			printf("%.2lf\n", a[i][n]);
		}
	}
	return 0;
}
