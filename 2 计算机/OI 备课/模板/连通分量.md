## 边双连通分量
```cpp
#include <iostream>
#include <cstring>
#include <cstdio>
#include <vector>

using namespace std;

const int N = 5e5 + 10, M = 4e6 + 10;

int n, m;
int h[N], e[M], ne[M], idx;
int dfn[N], low[N], timestamp;
int id[N], stk[N], top, dcc_cnt;
vector<int> ans[N];
bool is_bridge[M];
int tot[N];

int read()
{
	int s = 0, w = 1;
	char c = getchar();
	while (c < '0' || c > '9')
	{
		if (c == '-') w = -1;
		c = getchar();
	}
	while (c >= '0' && c <= '9')
	{
		s = s * 10 + c - '0';
		c = getchar();
	}
	return s * w;
}

void add(int a, int b)
{
	e[idx] = b;
	ne[idx] = h[a];
	h[a] = idx ++ ;
	return;
}

void tarjan(int u, int from)
{
	dfn[u] = low[u] = ++ timestamp;
	stk[ ++ top] = u;
	for (int i = h[u]; ~i; i = ne[i])
	{
		int ver = e[i];
		if (!dfn[ver])
		{
			tarjan(ver, i);
			low[u] = min(low[u], low[ver]);
			if (low[ver] > dfn[u]) is_bridge[i] = is_bridge[i ^ 1] = true;
		}
		else if (i != (1 ^ from)) low[u] = min(low[u], dfn[ver]);
	}

	if (dfn[u] == low[u])
	{
		int y;
		dcc_cnt ++ ;
		do
		{
			y = stk[top -- ];
			id[y] = dcc_cnt;
		} while (y != u);
	}
	return;
}

int main()
{
// 	freopen("in.txt", "r", stdin);
// 	freopen("out.txt", "w", stdout);
	n = read(), m = read();
	memset(h, -1, sizeof h);
	for (int i = 1; i <= m; i ++ )
	{
		int a = read(), b = read();
		add(a, b), add(b, a);
	}
	for (int i = 1; i <= n; i ++ )
		if (!dfn[i]) tarjan(i, -1);
	cout << dcc_cnt << endl;

	for (int i = 1; i <= n; i ++ )
		tot[id[i]] ++ , ans[id[i]].push_back(i);
	int p = 1;
	while (tot[p])
	{
		printf("%d ", tot[p]);
		for (int i : ans[p]) printf("%d ", i);
		p ++ ;
		printf("\n");
	}
	return 0;
}
```

## 点双连通分量
```cpp
#include <iostream>
#include <cstring>
#include <vector>

using namespace std;

const int N = 5e5 + 10, M = 5e6 + 10;

int n, m, root;
int h[N], e[M], ne[M], idx;
int dfn[N], low[N], timestamp;
int id[N], stk[N], top, dcc_cnt;
vector<int> dcc[N];
bool cut[N];

int read()
{
	int s = 0, w = 1;
	char c = getchar();
	while (c < '0' || c > '9')
	{
		if (c == '-') w = -1;
		c = getchar();
	}
	while (c >= '0' && c <= '9')
	{
		s = s * 10 + c - '0';
		c = getchar();
	}
	return s * w;
}

void add(int a, int b)
{
	e[idx] = b;
	ne[idx] = h[a];
	h[a] = idx ++ ;
	return;
}

void tarjan(int u)
{
	dfn[u] = low[u] = ++ timestamp;
	stk[ ++ top] = u;
	
	if (u == root && h[u] == -1)
	{
		dcc_cnt ++ ;
		dcc[dcc_cnt].push_back(u);
		return;
	}
	
	int child = 0;
	for (int i = h[u]; ~i; i = ne[i])
	{
		int ver = e[i];
		if (!dfn[ver])
		{
			tarjan(ver);
			low[u] = min(low[u], low[ver]);
			if (low[ver] >= dfn[u])
			{
				child ++ ;
				if (u != root || child >= 2) cut[u] = true;
				
				++ dcc_cnt;
				int y;
				do
				{
					y = stk[top -- ];
					dcc[dcc_cnt].push_back(y);
				} while (y != ver);
				dcc[dcc_cnt].push_back(u);
			}
		}
		else low[u] = min(low[u], dfn[ver]);
	}
	return;
}

int main()
{
	ios::sync_with_stdio(false);
	cin.tie(0);
	n = read(), m = read();
	memset(h, -1, sizeof h);
	for (int i = 1; i <= m; i ++ )
	{
		int a, b;
		a = read(), b = read();
		if (a == b) continue;
		add(a, b);
		add(b, a);
	}
	for (root = 1; root <= n; root ++ )
		if (!dfn[root]) tarjan(root);
	
	cout << dcc_cnt << endl;
	for (int i = 1; i <= dcc_cnt; i ++ )
	{
		cout << dcc[i].size() << ' ';
		for (int j = 0; j < dcc[i].size(); j ++ ) cout << dcc[i][j] << ' ';
		cout << endl;
	}
	return 0;
}
```

## 强连通分量
```cpp
#include <iostream>
#include <cstring>
#include <cstdio>

using namespace std;

const int N = 110, M = 10010;

int n;
int h[N], e[M], ne[M], idx;
int dfn[N], low[N], timestamp;
int stk[N], id[N], top, scc_cnt;
int din[N], dout[N];
bool in_stk[N];

void add(int a, int b)
{
    e[idx] = b;
    ne[idx] = h[a];
    h[a] = idx ++ ;
    return;
}

void tarjan(int u)
{
    dfn[u] = low[u] = ++ timestamp;
    stk[ ++ top] = u;
    in_stk[u] = true;

    for (int i = h[u]; ~i; i = ne[i])
    {
        int ver = e[i];
        if (!dfn[ver])
        {
            tarjan(ver);
            low[u] = min(low[u], low[ver]);
        }
        else if (in_stk[ver])
            low[u] = min(low[u], dfn[ver]);
    }

    if (dfn[u] == low[u])
    {
        scc_cnt ++ ;
        int y;
        do
        {
            y = stk[top -- ];
            in_stk[y] = false;
            id[y] = scc_cnt;
        } while (u != y);
    }

    return;
}

int main()
{
    // freopen("in.txt", "r", stdin);
    // freopen("out.txt", "w", stdout);
    cin >> n;
    memset(h, -1, sizeof h);
    for (int i = 1; i <= n; i ++ )
    {
        int t;
        while (cin >> t, t) add(i, t);
    }
    
    for (int i = 1; i <= n; i ++ )
        if (!dfn[i]) tarjan(i);
    for (int u = 1; u <= n; u ++ )
        for (int i = h[u]; ~i; i = ne[i])
        {
            int ver = e[i];
            int a = id[u], b = id[ver];
            if (a != b) dout[a] ++ , din[b] ++ ;
        }
    int a = 0, b = 0;
    for (int i = 1; i <= scc_cnt; i ++ )
    {
        if (!din[i]) a ++ ;
        if (!dout[i]) b ++ ;
    }
    cout << a << endl;
    if (scc_cnt == 1) cout << 0 << endl;
    else cout << max(a, b) << endl;
    return 0;
}
```
