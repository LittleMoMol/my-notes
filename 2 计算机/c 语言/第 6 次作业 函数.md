void get_ans(double a, double b, double c)
{
	double delta = b * b - 4 * a * c;
	double x1 = 0, x2 = 0;
	if (delta < 0)
	{
		printf("无解");
		return;
	}
	x1 = (-b + sqrt(delta)) / (2 * a);
	x2 = (-b - sqrt(delta)) / (2 * a);
	
	if (delta > 0) printf("两根为 %lf, %lf", x1, x2);
	else printf("有两个相同根 %lf", x1);
	return;
}
int main()
{
	double a, b, c;
	scanf("%lf %lf %lf", &a, &b, &c);
	get_ans(a, b, c);
	return 0;
}

void work(int a, int b, int c, int d)
{
	double L = -10, R = 10, mid = (L + R) / 2;
	while (R - L > 0.000001)
	{
		mid = (L + R) / 2;
		double val = a * mid * mid * mid + b * mid * mid + c * mid + d;
		if (val > 0) R = mid;
		else L = mid;
	}
	printf("%.2lf", mid);
	return;
}

int main()
{
	work(1, 2, 3, 4);
	return 0;

#define M 80
int fun(char a[M])
{
    int i, res = 0;
    while (a[i] != '\0')
    {
        if (a[i] >= 'a' && a[i] <= 'f')
            res = res * 16 + a[i] - 'a' + 10;
        if (a[i] >= 'A' && a[i] <= 'F')
            res = res * 16 + a[i] - 'A' + 10;
        if (a[i] >= '0' && a[i] <= '9')
            res = res * 16 + a[i] - '0';
        i++;
    }
    return res;
}
int main()
{
    char a[M];
    int i = 0, res;
    gets(a);
    while (a[i] != '\0')
    {
        if (!(a[i] >= '0' && a[i] <= '9' || a[i] >= 'a' && a[i] <= 'f' || a[i] >= 'A' && a[i] <= 'F'))
        {
            i = 0;
            printf("\n输入有误，请重新输入：");
            gets(a);
            continue;
        }
        i++;
    }
    res = fun(a);
    printf("\n( %s )16 到 ( %d )10\n", a, res);
    return 0;
}
