# 第 1 题
`1e-6` 时运行了 `500000` 次
`1e-8` 时运行了 `50000000` 次
代码如下（代码中的 `tot` 变量数值即为答案）：
```c
#include <stdio.h>
#include <math.h>

int main()
{
	int sign = 1, tot = 0;
	double pi = 0.0, n = 1.0, term = 1.0;
	
	while (fabs(term) >= 1e-6)
	{
		tot ++ ;
		pi = pi + term;
		n = n + 2;
		sign = -sign;
		term = sign / n;
	}
	pi = pi * 4;
	printf("pi=%10.8f\n", pi);
	printf("运行了 %d 次", tot);
	return 0;
}
```
以及

# 第 2 题
```c
#include <stdio.h>

int gcd(int a, int b) //求 a 和 b 的最大公约数 
{
	if (b == 0) return a;
	return gcd(b, a % b);
}

int lcm(int a, int b) //求 a 和 b 的最小公倍数。运用性质 a*b = gcd(a,b) * lcm(a,b) 
{
	return a * b / gcd(a, b);
}

int main()
{
	int m, n;
	scanf("%d %d", &m, &n);
	printf("最大公约数为：%d \n最小公倍数为：%d\n", gcd(m, n), lcm(m, n));
	return 0;
}
```

# 第 10 题
根据题意可知：球第 $i$ 次反弹有 $100 \times \left( \dfrac 12 \right)^i \mathrm{m}$ 高；第 $i$ 次落地时共经过 $100 + 2\sum\limits_{k=1}^{i-1} 100\times\left( \dfrac 12 \right)^k \mathrm{m}$ 
```c
#include <stdio.h>
#include <math.h>

double ans1, ans2;

int main()
{
	ans2 = 100 * pow(0.5, 10);
	ans1 = 100;
	for (int i = 1; i <= 9; i ++ ) ans1 += 2 * 100 * pow(0.5, i);
	printf("第 10 次落地时共经过 %.2lf 米\n", ans1);
	printf("第 10 次落地时反弹 %.2lf 米\n", ans2);
	return 0;
}
```

# 第 14 题
对函数求导可得 $y'=6x^2-8x+3$，恒大于零，故原函数单增
```c
#include <stdio.h>

double ans;

double f(double x)
{
	return 2 * x * x * x - 4 * x * x + 3 * x - 6;
}

int main()
{
	double L = -10.0, R = 10.0, mid = (L + R) / 2.0;
	while (R - L > 1e-6)
	{
		mid = (L + R) / 2.0;
		if (f(mid) > 0) R = mid, ans = mid;
		else L = mid, ans = mid;
	}
	printf("%.2lf", ans);
}
```
# 第 15 题
```c
#include <stdio.h>

int n;

int main()
{
	//输出菱形大小与 n 正相关（n 越大，菱形越大） 
	scanf("%d", &n);
	for (int i = 1; i <= n; i ++ )
	{
		for (int j = 1; j <= n - i; j ++ ) printf(" ");
		for (int j = 1; j <= 2 * i - 1; j ++ ) printf("*");
		printf("\n");
	}
	n -- ;
	for (int i = 1; i <= n; i ++ )
	{
		for (int j = 1; j <= i; j ++ ) printf(" ");
		for (int j = 1; j <= (n - i) * 2 + 1; j ++ ) printf("*");
		printf("\n");
	}
	return 0;
}
```
