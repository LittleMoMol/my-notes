void sort(int **q, int n)
{

	{
		for(j = 0; j < n - i - 1; j++)
		{
			if(*(*(q + j)) > *(*(q + j + 1)))
			{
				t = *(*(q + j));
				*(*(q + j)) = *(*(q + j + 1));
				*(*(q + j + 1)) = t;
			}
		}

}
