# 2 源代码
所有代码如下：
```sql
create table PAPER (  
    pno varchar(30) primary key comment '报纸编号',  
    pna varchar(30)             comment '报纸名称',  
    ppr decimal(5,2)            comment '单价'  
) comment '报纸编码表';  
  
create table CUSTOMER (  
    cno varchar(10) primary key comment '顾客编号',  
    cna varchar(30)             comment '顾客姓名',  
    adr varchar(90)             comment '顾客地址'  
) comment '顾客编码表';  
  
create table CP (  
    cno varchar(10)   comment '顾客编号',  
    pno varchar(30)   comment '报纸编号',  
    num int default 1 comment '订阅份数',  
    primary key (pno, cno),  -- 联合主键  
    foreign key (pno) references PAPER(pno),  
    foreign key (cno) references CUSTOMER(cno)  
) comment '顾客订阅表';

CREATE VIEW C_P_N AS  
SELECT  
    c.cno AS '顾客编号',  
    c.cna AS '顾客名称',  
    p.pno AS '报纸编号',  
    p.pna AS '报纸名称',  
    cp.num AS '订阅份数'  
FROM CP cp  
         JOIN CUSTOMER c ON cp.cno = c.cno  
         JOIN PAPER p ON cp.pno = p.pno;  
  
ALTER VIEW C_P_N AS  
    SELECT        c.cno AS '顾客编号',  
        c.cna AS '顾客名称',  
        p.pno AS '报纸编号',  
        p.pna AS '报纸名称',  
        cp.num AS '订阅份数',  
        p.ppr AS '报纸单价'  
    FROM CP cp  
             JOIN CUSTOMER c ON cp.cno = c.cno  
             JOIN PAPER p ON cp.pno = p.pno;  
  
SELECT * FROM C_P_N WHERE '报纸名称' = '人民日报';  
  
DROP VIEW C_P_N;
```
